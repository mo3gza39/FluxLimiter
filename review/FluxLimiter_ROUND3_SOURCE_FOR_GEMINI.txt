    private sealed record ProcessRow(string Name, int Pid, string Path, bool IsSystem);

    private void QueueReload()
    {
        if (IsDisposed) return;
        if (reloading) { refreshQueued = true; return; }
        _ = ReloadAsync();
    }

    private async Task ReloadAsync()
    {
        if (IsDisposed) return;
        if (reloading) { refreshQueued = true; return; }
        reloading = true;
        try
        {
            string query = search.Text.Trim();
            int filterIndex = filter.SelectedIndex;
            int? selectedPid = grid.CurrentRow?.Cells["pid"].Value is int pid ? pid : null;
            string? selectedPath = grid.CurrentRow?.Cells["path"].Value?.ToString();
            int firstVisible = grid.FirstDisplayedScrollingRowIndex;

            var snapshots = await Task.Run(() => SnapshotProcesses(query, filterIndex));
            if (IsDisposed) return;
            var rows = snapshots.Select(p =>
            {
                var rule = RuleCatalog.Find(settings, p.Path);
                return new object[] { p.Name, rule?.Blocked == true ? "Blocked" : "Active", "—", "—", "—",
                    rule?.DownloadLimitKBps?.ToString(CultureInfo.InvariantCulture) ?? "Unlimited",
                    rule?.UploadLimitKBps?.ToString(CultureInfo.InvariantCulture) ?? "Unlimited",
                    rule is null ? "None" : "Custom", "•••", p.Pid, p.Path };
            }).ToList();

            grid.SuspendLayout();
            try
            {
                grid.Rows.Clear();
                int selectedIndex = -1;
                for (int i = 0; i < rows.Count; i++)
                {
                    int row = grid.Rows.Add(rows[i]);
                    if (selectedPid.HasValue && Equals(rows[i][9], selectedPid.Value) &&
                        string.Equals(rows[i][10]?.ToString(), selectedPath, StringComparison.OrdinalIgnoreCase))
                        selectedIndex = row;
                }
                grid.ClearSelection();
                if (selectedIndex >= 0)
                {
                    grid.Rows[selectedIndex].Selected = true;
                    grid.CurrentCell = grid.Rows[selectedIndex].Cells[0];
                }
                if (firstVisible >= 0 && firstVisible < grid.Rows.Count)
                    grid.FirstDisplayedScrollingRowIndex = firstVisible;
            }
            finally { grid.ResumeLayout(); }
            status.Text = "Right-click an application for controls";
            processCount.Text = rows.Count.ToString(CultureInfo.InvariantCulture) + "  Processes";
            ruleCount.Text = settings.Rules.Count.ToString(CultureInfo.InvariantCulture) + "  Saved rules";
        }
        finally
        {
            reloading = false;
            if (refreshQueued && !IsDisposed)
            {
                refreshQueued = false;
                BeginInvoke((Action)QueueReload);
            }
        }
    }

    private static List<ProcessRow> SnapshotProcesses(string query, int filterIndex)
    {
        string windows = Environment.GetFolderPath(Environment.SpecialFolder.Windows);
        var rows = new List<ProcessRow>();
        foreach (var p in Process.GetProcesses().OrderBy(p => p.ProcessName))
        {
            try
            {
                string name = p.ProcessName;
                string path;
                try { path = p.MainModule?.FileName ?? ""; } catch { path = ""; }
                if (!name.Contains(query, StringComparison.OrdinalIgnoreCase) && !path.Contains(query, StringComparison.OrdinalIgnoreCase)) continue;
                bool isSystem = path.StartsWith(windows, StringComparison.OrdinalIgnoreCase);
                if (filterIndex == 1 && isSystem) continue;
                if (filterIndex == 2 && !isSystem) continue;
                rows.Add(new ProcessRow(name, p.Id, path, isSystem));
            }
            catch (InvalidOperationException) { }
            catch (System.ComponentModel.Win32Exception) { }
            finally { p.Dispose(); }
        }
        return rows;
    }

    private void Reload() => QueueReload();

    private void SaveInlineRule(int rowIndex)
    {
        string path = grid.Rows[rowIndex].Cells["path"].Value?.ToString() ?? "";
        if (string.IsNullOrWhiteSpace(path)) return;
        if (!TryLimit(grid.Rows[rowIndex].Cells["down"].Value?.ToString() ?? "", out var down) ||
            !TryLimit(grid.Rows[rowIndex].Cells["up"].Value?.ToString() ?? "", out var up))
        {
            MessageBox.Show("Use a positive whole number in KB/s, or Unlimited.", "Invalid limit");
            Reload();
            return;
        }
        var rule = RuleCatalog.Find(settings, path) ?? new AppRule { ExecutablePath = path };
        rule.DownloadLimitKBps = down;
        rule.UploadLimitKBps = up;
        if (!settings.Rules.Contains(rule)) settings.Rules.Add(rule);
        if (!TrySaveSettings()) return;
        Reload();
    }

    private void EditRule(int index)
    {
        string path = grid.Rows[index].Cells["path"].Value?.ToString() ?? "";
        if (string.IsNullOrWhiteSpace(path))
        {
            MessageBox.Show("Executable path unavailable. Try running FluxLimiter as administrator.", "FluxLimiter");
            return;
        }
        var rule = settings.Rules.FirstOrDefault(r => string.Equals(r.ExecutablePath, path, StringComparison.OrdinalIgnoreCase));
        using var dialog = new Form { Text = "Saved rule — " + Path.GetFileName(path), Width = 450, Height = 260, StartPosition = FormStartPosition.CenterParent, FormBorderStyle = FormBorderStyle.FixedDialog, MaximizeBox = false, MinimizeBox = false, BackColor = panel, ForeColor = ink, Font = Font };
        var dl = new TextBox { Left = 170, Top = 25, Width = 225, Text = rule?.DownloadLimitKBps?.ToString() ?? "", BackColor = bg, ForeColor = ink };
        var ul = new TextBox { Left = 170, Top = 65, Width = 225, Text = rule?.UploadLimitKBps?.ToString() ?? "", BackColor = bg, ForeColor = ink };
        var blocked = new CheckBox { Left = 170, Top = 105, Text = "Block application", Checked = rule?.Blocked ?? false, AutoSize = true };
        dialog.Controls.AddRange(new Control[] {
            new Label { Text = "Download KB/s", Left = 18, Top = 28, AutoSize = true },
            new Label { Text = "Upload KB/s", Left = 18, Top = 68, AutoSize = true },
            dl, ul, blocked,
            new Label { Text = "Blank = unlimited. Rules saved but not enforced in this preview.", Left = 18, Top = 143, Width = 410, Height = 34 }
        });
        var save = new Button { Text = "Save rule", Left = 280, Top = 176, Width = 115, DialogResult = DialogResult.OK, BackColor = Color.FromArgb(45, 74, 91), ForeColor = Color.White, FlatStyle = FlatStyle.Flat };
        dialog.Controls.Add(save);
        dialog.AcceptButton = save;
        if (dialog.ShowDialog(this) != DialogResult.OK) return;
        if (!TryLimit(dl.Text, out var down) || !TryLimit(ul.Text, out var up))
        {
            MessageBox.Show("Enter a positive whole number, or leave blank for unlimited.", "Invalid limit");
            return;
        }
        rule ??= new AppRule { ExecutablePath = path };
        rule.DownloadLimitKBps = down;
        rule.UploadLimitKBps = up;
        rule.Blocked = blocked.Checked;
        if (!settings.Rules.Contains(rule)) settings.Rules.Add(rule);
        if (!TrySaveSettings()) return;
        Reload();
    }

    private static bool TryLimit(string value, out int? limit)
    {
        limit = null;
        if (string.IsNullOrWhiteSpace(value) || value.Trim().Equals("Unlimited", StringComparison.OrdinalIgnoreCase)) return true;
        if (!int.TryParse(value, out int n) || n <= 0) return false;
        limit = n;
        return true;
    }

    private void ToggleBlockSelected()
    {
        string path = grid.CurrentRow?.Cells["path"].Value?.ToString() ?? "";
        if (string.IsNullOrWhiteSpace(path)) { MessageBox.Show("Executable path unavailable."); return; }
        var rule = RuleCatalog.Find(settings, path);
        if (rule is null)
        {
            rule = new AppRule { ExecutablePath = path, Blocked = true };
            settings.Rules.Add(rule);
        }
        else rule.Blocked = !rule.Blocked;
        if (!TrySaveSettings()) return;
        Reload();
    }

    private bool TrySaveSettings()
    {
        if (settingsSaveDisabled)
        {
            MessageBox.Show("Settings cannot be saved because the previous settings file could not be safely preserved.", "Saving disabled");
            return false;
        }
        try
        {
            SettingsStore.Save(settings);
            return true;
        }
        catch (Exception ex) when (ex is IOException or UnauthorizedAccessException)
        {
            MessageBox.Show(ex.Message, "Unable to save settings");
            return false;
        }
    }

using System.Diagnostics;
using System.Net.NetworkInformation;

namespace FluxLimiter;

// Device-level adapter throughput only. Does not attribute traffic to applications.
public sealed class NetworkSampler
{
    private readonly Dictionary<string, (long Received, long Sent)> previous = new();
    private long lastTick;
    public (double DownloadKBps, double UploadKBps) Sample()
    {
        long now = Stopwatch.GetTimestamp();
        double elapsed = lastTick == 0 ? 0 : (now - lastTick) / (double)Stopwatch.Frequency;
        double down = 0, up = 0;
        var live = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        foreach (var adapter in NetworkInterface.GetAllNetworkInterfaces())
        {
            if (adapter.OperationalStatus != OperationalStatus.Up ||
                adapter.NetworkInterfaceType == NetworkInterfaceType.Loopback ||
                adapter.NetworkInterfaceType == NetworkInterfaceType.Tunnel) continue;
            try
            {
                var stats = adapter.GetIPv4Statistics();
                string id = adapter.Id;
                live.Add(id);
                if (elapsed > 0 && previous.TryGetValue(id, out var old))
                {
                    long received = stats.BytesReceived - old.Received;
                    long sent = stats.BytesSent - old.Sent;
                    if (received >= 0) down += received;
                    if (sent >= 0) up += sent;
                }
                previous[id] = (stats.BytesReceived, stats.BytesSent);
            }
            catch (NetworkInformationException) { }
        }
        foreach (string id in previous.Keys.Where(k => !live.Contains(k)).ToArray()) previous.Remove(id);
        lastTick = now;
        return elapsed > 0 ? (down / elapsed / 1024.0, up / elapsed / 1024.0) : (0, 0);
    }
}

// Configuration model only. Network enforcement is NOT implemented in this prototype.
public static class RuleCatalog
{
    public static string NormalizePath(string path)
    {
        if (string.IsNullOrWhiteSpace(path)) return "";
        try { return Path.GetFullPath(path.Trim().Trim('"')); }
        catch (ArgumentException) { return ""; }
        catch (NotSupportedException) { return ""; }
        catch (PathTooLongException) { return ""; }
    }

    public static AppRule? Find(AppSettings settings, string executablePath)
    {
        string normalized = NormalizePath(executablePath);
        if (normalized.Length == 0) return null;
        return settings.Rules.FirstOrDefault(r =>
            string.Equals(NormalizePath(r.ExecutablePath), normalized, StringComparison.OrdinalIgnoreCase));
    }

    public static (int? DownKBps, int? UpKBps, bool Blocked, string Scope) Resolve(AppSettings settings, string executablePath)
    {
        var rule = Find(settings, executablePath);
        if (rule is null) return (null, null, false, "Unmanaged");
        var group = settings.Groups.FirstOrDefault(g => !string.IsNullOrWhiteSpace(rule.GroupId) &&
            string.Equals(g.Id, rule.GroupId, StringComparison.OrdinalIgnoreCase));
        return group is null
            ? (rule.DownloadLimitKBps, rule.UploadLimitKBps, rule.Blocked, "Executable")
            : (group.DownloadLimitKBps, group.UploadLimitKBps, rule.Blocked, "Group: " + group.Name);
    }

    public static bool ValidLimit(int? value) => value is null || value > 0;

    public static string ProcessKey(int pid, string path) => pid.ToString(System.Globalization.CultureInfo.InvariantCulture) + "|" + NormalizePath(path);
}

using System.Text.Json;

namespace FluxLimiter;

public static class SettingsStore
{
    private static readonly string Folder = Path.Combine(
        Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "FluxLimiter");
    private static readonly string FilePath = Path.Combine(Folder, "settings.json");
    private static readonly JsonSerializerOptions JsonOptions = new() { WriteIndented = true };
    public static string? LastLoadWarning { get; private set; }

    public static AppSettings Load()
    {
        LastLoadWarning = null;
        try
        {
            if (!File.Exists(FilePath)) return new();
            var settings = JsonSerializer.Deserialize<AppSettings>(File.ReadAllText(FilePath)) ?? new();
            settings.Rules ??= new();
            settings.Groups ??= new();
            // Ignore invalid/duplicate paths and invalid limits, rather than applying ambiguous rules.
            var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            settings.Rules = settings.Rules.Where(r =>
                !string.IsNullOrWhiteSpace(RuleCatalog.NormalizePath(r.ExecutablePath)) &&
                RuleCatalog.ValidLimit(r.DownloadLimitKBps) && RuleCatalog.ValidLimit(r.UploadLimitKBps) &&
                seen.Add(RuleCatalog.NormalizePath(r.ExecutablePath))).ToList();
            settings.Groups = settings.Groups.Where(g =>
                !string.IsNullOrWhiteSpace(g.Id) &&
                RuleCatalog.ValidLimit(g.DownloadLimitKBps) && RuleCatalog.ValidLimit(g.UploadLimitKBps))
                .GroupBy(g => g.Id, StringComparer.OrdinalIgnoreCase).Select(g => g.First()).ToList();
            return settings;
        }
        catch (Exception ex) when (ex is JsonException or IOException or UnauthorizedAccessException)
        {
            // Never leave a corrupt primary file in place where a later Save could replace it.
            try
            {
                if (File.Exists(FilePath))
                {
                    string corrupt = FilePath + ".corrupt-" + DateTime.UtcNow.ToString("yyyyMMdd-HHmmss", System.Globalization.CultureInfo.InvariantCulture);
                    File.Move(FilePath, corrupt, overwrite: false);
                    LastLoadWarning = "Settings could not be loaded. The original file was preserved as " + Path.GetFileName(corrupt) + ".";
                }
                else LastLoadWarning = "Settings could not be loaded: " + ex.Message;
            }
            catch (Exception preserveEx) when (preserveEx is IOException or UnauthorizedAccessException)
            {
                // Refuse silent destructive recovery if the original cannot be preserved.
                LastLoadWarning = "Settings could not be loaded or preserved. Saving is disabled for this session.";
            }
            return new();
        }
    }

    public static void Save(AppSettings settings)
    {
        Directory.CreateDirectory(Folder);
        string temp = FilePath + ".tmp";
        string backup = FilePath + ".bak";
        File.WriteAllText(temp, JsonSerializer.Serialize(settings, JsonOptions));
        try
        {
            if (File.Exists(FilePath))
                File.Replace(temp, FilePath, backup);
            else
                File.Move(temp, FilePath);
        }
        finally
        {
            if (File.Exists(temp)) File.Delete(temp);
        }
    }
}

namespace FluxLimiter;

// Pure, deterministic scheduler core. It does NOT intercept or throttle OS traffic.
// All byte budgets are binary KB/s (1 KB = 1024 bytes).
public sealed class SharedRateLimiter
{
    private sealed class Bucket
    {
        public double Tokens;
        public double LastSeconds;
        public double Capacity;
        public double Rate;
    }

    private readonly Dictionary<string, Bucket> buckets = new(StringComparer.OrdinalIgnoreCase);
    private readonly object gate = new();

    public bool TryConsume(string identity, int limitKBps, int bytes, double monotonicSeconds)
    {
        if (string.IsNullOrWhiteSpace(identity)) throw new ArgumentException("Identity is required.", nameof(identity));
        if (limitKBps <= 0) throw new ArgumentOutOfRangeException(nameof(limitKBps));
        if (bytes < 0) throw new ArgumentOutOfRangeException(nameof(bytes));
        if (double.IsNaN(monotonicSeconds) || double.IsInfinity(monotonicSeconds) || monotonicSeconds < 0)
            throw new ArgumentOutOfRangeException(nameof(monotonicSeconds));
        double rate = limitKBps * 1024.0;
        lock (gate)
        {
            if (!buckets.TryGetValue(identity, out var bucket))
            {
                bucket = new Bucket { Rate = rate, Capacity = rate, Tokens = rate, LastSeconds = monotonicSeconds };
                buckets.Add(identity, bucket);
            }
            else
            {
                double elapsed = Math.Max(0, monotonicSeconds - bucket.LastSeconds);
                bucket.Tokens = Math.Min(bucket.Capacity, bucket.Tokens + elapsed * bucket.Rate);
                bucket.LastSeconds = Math.Max(bucket.LastSeconds, monotonicSeconds);
                if (bucket.Rate != rate)
                {
                    bucket.Rate = rate;
                    bucket.Capacity = rate;
                    bucket.Tokens = Math.Min(bucket.Tokens, rate);
                }
            }
            if (bytes > bucket.Tokens) return false;
            bucket.Tokens -= bytes;
            return true;
        }
    }

    // The caller must split payloads larger than the one-second bucket capacity.
    // Direction-specific keys keep download and upload accounting independent.
    public static string DirectionKey(string scope, bool download)
    {
        if (string.IsNullOrWhiteSpace(scope)) throw new ArgumentException("Scope is required.", nameof(scope));
        return (download ? "down:" : "up:") + scope.Trim();
    }

    public void Remove(string identity)
    {
        lock (gate) buckets.Remove(identity);
    }
}

using System;
using System.Collections.Generic;

namespace FluxLimiter;

/// <summary>
/// Deterministic simulation-only byte queue. It does not intercept operating-system traffic.
/// Each identity and direction has an independent token bucket.
/// </summary>
public sealed class TrafficPacer
{
    private sealed class Budget
    {
        public double Tokens;
        public double UpdatedAt;
        public double Rate;
    }

    private readonly Dictionary<string, Budget> budgets = new(StringComparer.OrdinalIgnoreCase);
    private readonly object sync = new();

    /// <summary>Returns the earliest permitted monotonic time, in seconds, for a byte chunk.</summary>
    public double Schedule(string identity, bool download, int limitKBps, int bytes, double nowSeconds)
    {
        if (string.IsNullOrWhiteSpace(identity)) throw new ArgumentException("Required", nameof(identity));
        if (limitKBps <= 0) throw new ArgumentOutOfRangeException(nameof(limitKBps));
        if (bytes < 0) throw new ArgumentOutOfRangeException(nameof(bytes));
        if (!double.IsFinite(nowSeconds) || nowSeconds < 0) throw new ArgumentOutOfRangeException(nameof(nowSeconds));
        double rate = limitKBps * 1024.0;
        string key = SharedRateLimiter.DirectionKey(identity, download);
        lock (sync)
        {
            if (!budgets.TryGetValue(key, out var budget))
            {
                budget = new Budget { Tokens = rate, Rate = rate, UpdatedAt = nowSeconds };
                budgets.Add(key, budget);
            }
            double effectiveNow = Math.Max(nowSeconds, budget.UpdatedAt);
            if (budget.Rate != rate)
            {
                budget.Rate = rate;
                budget.Tokens = Math.Min(budget.Tokens, rate);
            }
            budget.Tokens = Math.Min(rate, budget.Tokens + (effectiveNow - budget.UpdatedAt) * rate);
            budget.UpdatedAt = effectiveNow;
            // Negative tokens represent queued work. Large requests are paced rather than denied forever.
            budget.Tokens -= bytes;
            return budget.Tokens >= 0 ? effectiveNow : effectiveNow - budget.Tokens / rate;
        }
    }

    public void Clear(string identity)
    {
        lock (sync)
        {
            budgets.Remove(SharedRateLimiter.DirectionKey(identity, true));
            budgets.Remove(SharedRateLimiter.DirectionKey(identity, false));
        }
    }
}
